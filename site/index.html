TypeScript · Edge Auth Filterfilter.ts

import { verifyHmac } from "envoy:crypto/hmac@0.3.0";
import { checkQuota } from "envoy:ratelimit/quota@0.3.0";

export async function onRequestHeaders(
  token: string,
  route: string
): Promise<{ allow: boolean; tenant: string; status: number }> {
  const claims = verifyHmac(token);
  if (!claims.valid) return { allow: false, tenant: "", status: 401 };
  const remaining = await checkQuota(claims.tenant, route);
  return {
    allow: remaining > 0,
    tenant: claims.tenant,
    status: remaining > 0 ? 200 : 429,
  };
}

    tenant: string,

    token: string,
    route: string
  ) -> decision;
